cmake_minimum_required(VERSION 3.22)
project(onero_a1_kinematics LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
find_package(ament_cmake REQUIRED)
find_package(ament_cmake_python REQUIRED)
find_package(ament_index_cpp REQUIRED)
find_package(Eigen3 REQUIRED)
find_package(moveit_core REQUIRED)
find_package(moveit_kinematics REQUIRED)
find_package(pluginlib REQUIRED)
find_package(rclcpp REQUIRED)
find_package(urdf REQUIRED)
find_package(srdfdom REQUIRED)
find_package(yaml_cpp_vendor REQUIRED)
find_package(yaml-cpp REQUIRED)
find_package(pybind11_vendor REQUIRED)
find_package(Python3 REQUIRED COMPONENTS Interpreter Development)
find_package(pybind11 REQUIRED)
find_library(KDL_PLUGIN_LIBRARY moveit_kdl_kinematics_plugin REQUIRED)

add_library(onero_posture_core SHARED src/profile.cpp)
target_include_directories(onero_posture_core PUBLIC $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include>
                                                    $<INSTALL_INTERFACE:include>)
target_link_libraries(onero_posture_core Eigen3::Eigen yaml-cpp)
ament_target_dependencies(onero_posture_core ament_index_cpp)

pybind11_add_module(_core src/python.cpp src/native_planning.cpp)
target_link_libraries(_core PRIVATE onero_posture_core)
ament_target_dependencies(_core PUBLIC moveit_core rclcpp urdf srdfdom)
ament_python_install_package(${PROJECT_NAME})
ament_get_python_install_dir(PYTHON_INSTALL_DIR)
install(TARGETS _core LIBRARY DESTINATION "${PYTHON_INSTALL_DIR}/${PROJECT_NAME}")

add_library(onero_posture_ik SHARED src/ik_plugin.cpp)
target_link_libraries(onero_posture_ik onero_posture_core ${KDL_PLUGIN_LIBRARY})
ament_target_dependencies(onero_posture_ik moveit_core moveit_kinematics pluginlib rclcpp)
pluginlib_export_plugin_description_file(moveit_core kinematics_plugin.xml)
install(TARGETS onero_posture_core EXPORT onero_postureTargets LIBRARY DESTINATION lib)
install(TARGETS onero_posture_ik LIBRARY DESTINATION lib)
install(DIRECTORY include/ DESTINATION include)
install(DIRECTORY config DESTINATION share/${PROJECT_NAME})
ament_export_include_directories(include)
ament_export_targets(onero_postureTargets HAS_LIBRARY_TARGET)
ament_export_dependencies(Eigen3 ament_index_cpp yaml_cpp_vendor yaml-cpp)
ament_package()
